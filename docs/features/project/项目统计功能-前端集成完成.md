**核心代码**:

```typescript
import * as echarts from 'echarts'
import type { ECharts } from 'echarts'

export default function ProjectStatsChart() {
  const chartRef = useRef<HTMLDivElement>(null)
  const chartInstance = useRef<ECharts | null>(null)
  
  // 初始化图表
  useEffect(() => {
    if (chartRef.current && !chartInstance.current) {
      chartInstance.current = echarts.init(chartRef.current)
      
      // 监听窗口大小变化
      const handleResize = () => {
        chartInstance.current?.resize()
      }
      window.addEventListener('resize', handleResize)
      
      return () => {
        window.removeEventListener('resize', handleResize)
        chartInstance.current?.dispose()
      }
    }
  }, [])
  
  // 渲染图表
  const renderChart = (statsData) => {
    const option = {
      title: { text: '项目账号更新趋势' },
      tooltip: { trigger: 'axis' },
      legend: { data: statsData.map(item => item.project_name) },
      xAxis: { type: 'category', data: statsData[0]?.dates || [] },
      yAxis: { type: 'value', name: '更新数量' },
      series: statsData.map(item => ({
        name: item.project_name,
        type: 'line',
        data: item.counts,
        smooth: true
      }))
    }
    
    chartInstance.current.setOption(option, true)
  }
  
  return (
    <Card title="项目账号更新趋势" extra={/* 操作按钮 */}>
      <div ref={chartRef} style={{ width: '100%', height: 400 }} />
    </Card>
  )
}
```

// 获取仪表盘统计数据
export const getProjectStatsForDashboard = (params: { days: number }) => {
  return api.get('/v1/project/stats/dashboard', { params })
}

// 获取项目今天的更新数量
export const getProjectTodayCount = (projectId: string) => {
  return api.get(`/v1/project/stats/project/${projectId}/today`)
}

// 清除统计缓存（仅管理员）
export const clearStatsCache = (projectId?: string) => {
  return api.post('/v1/project/stats/cache/clear', null, {
    params: projectId ? { project_id: projectId } : undefined
  })
}

// 手动同步统计数据（仅管理员）
export const syncStatsData = (days: number = 1) => {
  return api.post('/v1/project/stats/sync', null, {
    params: { days }
  })
}

## 功能特性

### 1. 图表展示

- **曲线图**: 使用ECharts绘制平滑曲线
- **多项目**: 支持同时显示多个项目的数据
- **图例**: 可点击图例显示/隐藏对应项目
- **提示框**: 鼠标悬停显示详细数据
- **响应式**: 自动适应窗口大小

### 2. 时间范围

支持4种时间范围：

- 最近7天
- 最近14天
- 最近30天
- 最近90天

### 3. 操作功能

- **刷新**: 重新加载数据
- **清除缓存**: 管理员可清除Redis缓存
- **自动加载**: 页面加载时自动获取数据
- **切换范围**: 切换时间范围时自动重新加载

### 4. 权限控制

#### 数据权限
- **ADMIN/GM**: 看到所有项目的统计数据
- **IT/MANUAL**: 只看到分配给自己的项目

#### 功能权限
- **清除缓存按钮**: 仅ADMIN可见
- **同步数据API**: 仅ADMIN可调用

### 5. 用户体验

- **加载状态**: 显示Spin加载动画
- **空数据提示**: 无数据时显示友好提示
- **错误处理**: 网络错误时显示错误消息
- **性能优化**: 图表实例复用，避免重复创建

## 技术实现

### 1. ECharts配置

```typescript
const option: echarts.EChartsOption = {
  title: {
    text: '项目账号更新趋势',
    left: 'center',
    textStyle: {
      fontSize: 16,
      fontWeight: 'normal'
    }
  },
  tooltip: {
    trigger: 'axis',
    axisPointer: {
      type: 'cross',
      label: {
        backgroundColor: '#6a7985'
      }
    },
    formatter: (params: any) => {
      // 自定义提示框内容
      let result = `<div style="font-weight: bold;">${params[0].axisValue}</div>`
      params.forEach((param: any) => {
        result += `
          <div>
            ${param.marker}
            <span>${param.seriesName}</span>
            <span style="font-weight: bold;">${param.value} 个</span>
          </div>
        `
      })
      return result
    }
  },
  legend: {
    data: statsData.map(item => item.project_name),
    top: 35,
    type: 'scroll',
    pageButtonPosition: 'end'
  },
  grid: {
    left: '3%',
    right: '4%',
    bottom: '3%',
    containLabel: true
  },
  xAxis: {
    type: 'category',
    boundaryGap: false,
    data: statsData[0]?.dates || [],
    axisLabel: {
      rotate: 45,
      formatter: (value: string) => {
        const date = new Date(value)
        return `${date.getMonth() + 1}/${date.getDate()}`
      }
    }
  },
  yAxis: {
    type: 'value',
    name: '更新数量',
    minInterval: 1,
    axisLabel: {
      formatter: '{value} 个'
    }
  },
  series: statsData.map(item => ({
    name: item.project_name,
    type: 'line',
    data: item.counts,
    smooth: true,
    emphasis: {
      focus: 'series'
    },
    lineStyle: {
      width: 2
    },
    showSymbol: true,
    symbolSize: 6
  }))
}
```
